/* ---------- Loan: payment number generation & UI ---------- */
// create or read payment number from localStorage
function makeRandomPaymentNumber(){
  // generate like PAY-<timestamp>-<4digits>
  const ts = Date.now().toString().slice(-6);
  const rnd = Math.floor(Math.random()*9000)+1000;
  return `PAY-${ts}-${rnd}`;
}
let paymentNumber = localStorage.getItem('rc_payment_no');
if(!paymentNumber){
  paymentNumber = makeRandomPaymentNumber();
  localStorage.setItem('rc_payment_no', paymentNumber);
}
document.getElementById('payNumber').textContent = paymentNumber;
document.getElementById('payNote').textContent = 'Segera bayar';

/* copy to clipboard */
function copyPaymentNumber(){
  const txt = document.getElementById('payNumber').textContent;
  if(navigator.clipboard && navigator.clipboard.writeText){
    navigator.clipboard.writeText(txt).then(()=>{
      alert('Nomor pembayaran disalin: ' + txt);
    }).catch(()=>{ fallbackCopy(txt); });
  } else {
    fallbackCopy(txt);
  }
}
function fallbackCopy(text){
  const ta = document.createElement('textarea');
  ta.value = text; document.body.appendChild(ta); ta.select();
  try{
    document.execCommand('copy');
    alert('Nomor pembayaran disalin: ' + text);
  }catch(e){ prompt('Salin manual nomor pembayaran:', text); }
  ta.remove();
}

/* regenerate payment number (user requested "nomer terserah") */
function regeneratePayment(){
  if(confirm('Ganti nomor pembayaran dengan nomor baru?')){
    paymentNumber = makeRandomPaymentNumber();
    localStorage.setItem('rc_payment_no', paymentNumber);
    document.getElementById('payNumber').textContent = paymentNumber;
    alert('Nomor pembayaran baru: ' + paymentNumber);
  }
}

/* ---------- RC Poin logic (simple) ---------- */
let points = Number(localStorage.getItem('rc_points') || 88);
let streak = Number(localStorage.getItem('rc_streak') || 1);
let lastCheck = localStorage.getItem('rc_lastcheck') || null;
const pointSchedule = [10,10,10,20,10,10,30];

function renderPoints(){
  document.getElementById('pointsNum').textContent = points;
  document.getElementById('streak').textContent = (streak>7?7:streak);
  const grid = document.getElementById('checkGrid');
  grid.innerHTML = '';
  for(let i=0;i<7;i++){
    const day = i+1;
    const box = document.createElement('div');
    box.className = 'day-box' + (day <= streak ? ' active' : '');
    box.innerHTML = `<div style="font-size:12px;color:var(--muted)">Hari ke ${day}</div><div style="margin-top:6px;font-size:16px">+${pointSchedule[i]}</div>`;
    grid.appendChild(box);
  }
}
renderPoints();

function doCheckIn(){
  const today = new Date().toDateString();
  if(lastCheck === today){
    alert("Anda sudah Check-in hari ini.");
    return;
  }
  const dayIndex = Math.min(Math.max(streak-1,0),6);
  const reward = pointSchedule[dayIndex];
  points += reward;
  // determine streak continuity
  const yesterday = new Date();
  yesterday.setDate(new Date().getDate()-1);
  if(lastCheck === yesterday.toDateString()){
    streak = Math.min(streak + 1, 7);
  } else {
    streak = 2; // after first check-in, prepare next day; we use this to simulate progression
  }
  lastCheck = today;
  localStorage.setItem('rc_lastcheck', lastCheck);
  localStorage.setItem('rc_points', points);
  localStorage.setItem('rc_streak', streak);
  renderPoints();
  alert(`Check-in berhasil! Anda mendapatkan +${reward} poin.`);
}

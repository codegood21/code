@app.route("/contact", methods=["GET", "POST"])
def contact():
    if request.method == "POST":
        name = request.form.get("name","").strip()
        email = request.form.get("email","").strip()
        message = request.form.get("message","").strip()
        if not name or not message:
            flash("Isi nama dan pesan dulu ya.")
            return redirect(url_for("contact"))
        # Simpan sederhana ke file lokal (untuk demo). Ganti dengan DB atau email di produksi.
        entry = f"{datetime.utcnow().isoformat()} | {name} | {email} | {message}\\n"
        with open(CONTACT_STORAGE, "a", encoding="utf-8") as f:
            f.write(entry)
        flash("Pesan terkirim. Terima kasih!")
        return redirect(url_for("contact"))

    body = """
    <h2>Contact</h2>
    <form method="post" class="mt-3" style="max-width:720px">
      <div class="mb-3">
        <label class="form-label">Nama</label>
        <input name="name" class="form-control" placeholder="Nama kamu">
      </div>
      <div class="mb-3">
        <label class="form-label">Email (opsional)</label>
        <input name="email" class="form-control" placeholder="email@contoh.com">
      </div>
      <div class="mb-3">
        <label class="form-label">Pesan</label>
        <textarea name="message" rows="5" class="form-control" placeholder="Tinggalkan pesan..."></textarea>
      </div>
      <button class="btn btn-primary">Kirim</button>
    </form>
    <hr>
    <p class="small text-muted">Catatan: pesan tersimpan secara lokal di file <code>contacts.txt</code>. Untuk produksi, sambungkan SMTP atau database.</p>
    """
    return render_template_string(BASE_HTML, title="Contact", site=SITE, body=body, now=datetime.utcnow())

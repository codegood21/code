<script>
// --- CORE DATA & LOCALSTORAGE ---
let mode = localStorage.getItem('modeRT') || 'bendahara';
let dbData = JSON.parse(localStorage.getItem('rt_db')) || { pengumuman: "Selamat Datang di Sistem Smart RT!" };
let customTheme = JSON.parse(localStorage.getItem('rt_custom_theme')) || { accent: '#38bdf8', sidebar: '', dashboard: '' };

// Struktur data keuangan tetap menggunakan riwayat dan saldo
let dataRT = JSON.parse(localStorage.getItem('rt_keuangan')) || {
    kas: { saldo: 0, t_masuk: 0, t_keluar: 0, riwayat: [] },
    tabungan: { warga: [], saldo: [], riwayat: [], t_masuk: 0, t_keluar: 0 },
    arsip: { kk: [], domisili: [], dokumentasi: [] }
};

// --- NAVIGATION ---
function toggleSidebar(f){ 
    if(f) document.getElementById('bodyApp').classList.add('sidebar-open');
    else document.getElementById('bodyApp').classList.remove('sidebar-open');
}
function toggleMenu(id){ document.getElementById(id).classList.toggle('show'); }
function nav(id){
    document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
    document.getElementById(id).classList.add('active');
    document.getElementById('judul').innerText = id.toUpperCase();
    toggleSidebar(false);
    if(id==='db') renderCharts();
    render();
}

// --- MAIN RENDER ---
function render(){
    document.getElementById('db-pengumuman').innerText = dbData.pengumuman;
    document.getElementById('modeLabel').innerText = "MODE: " + mode.toUpperCase();

    // 1. RENDER KAS RT
    const uiKas = document.getElementById('content-kas');
    if(uiKas){
        uiKas.innerHTML = `
            <div class="balance-card">
                <small>Saldo Kas RT</small>
                <h2>Rp ${dataRT.kas.saldo.toLocaleString()}</h2>
                <div class="balance-stats">
                    <div>Pemasukan<br><b>Rp ${dataRT.kas.t_masuk.toLocaleString()}</b></div>
                    <div>Pengeluaran<br><b>Rp ${dataRT.kas.t_keluar.toLocaleString()}</b></div>
                </div>
            </div>
            ${mode==='bendahara'?`
            <div class="action-grid">
                <button class="btn-action bg-pemasukan" onclick="transKas('in')">➕ Pemasukan</button>
                <button class="btn-action bg-pengeluaran" onclick="transKas('out')">➖ Pengeluaran</button>
            </div>`:''}
            <div class="card">
                <b>📜 Riwayat Kas</b>
                <div style="max-height:250px;overflow-y:auto;margin-top:10px">
                    ${dataRT.kas.riwayat.length === 0 ? '<small style="opacity:0.5">Belum ada riwayat.</small>' : 
                    dataRT.kas.riwayat.slice().reverse().map(r=>`
                    <div class="ri-item-new">
                        <span><b style="color:${r.tipe==='in'?'#10b981':'#ef4444'}">${r.tipe==='in'?'[+]':'[-]'}</b> ${r.nama}</span>
                        <b>${r.nom.toLocaleString()}</b>
                    </div>`).join('')}
                </div>
                ${mode==='bendahara' ? `<button onclick="resetLog('kas')" style="width:100%;margin-top:10px;color:red;border:none;background:none;font-size:11px;cursor:pointer">Hapus Riwayat Saja (Saldo Tetap)</button>` : ''}
            </div>`;
    }

    // 2. RENDER TABUNGAN
    const uiTab = document.getElementById('content-tabungan');
    if(uiTab){
        let kw = (document.getElementById('searchTab')?.value || "").toUpperCase();
        let totalS = dataRT.tabungan.saldo.reduce((a,b)=>a+b,0);
        uiTab.innerHTML = `
            <div class="balance-card">
                <small>Total Tabungan Warga</small>
                <h2>Rp ${totalS.toLocaleString()}</h2>
            </div>
            <div class="card">
                <div style="display:flex;justify-content:space-between;align-items:center">
                    <b>👥 Daftar Warga</b> 
                    ${mode==='bendahara' ? `<button onclick="tambahWarga()" style="background:var(--accent-color);border:none;color:white;border-radius:4px;padding:2px 8px;cursor:pointer">+</button>` : ''}
                </div>
                <div class="table-container" style="margin-top:10px">
                    <table>
                        <tr><th>Nama Warga</th><th>Saldo</th><th style="text-align:center">Aksi</th></tr>
                        ${dataRT.tabungan.warga.map((w,i)=> w.includes(kw)? `
                        <tr>
                            <td>${w} ${mode==='bendahara'?`<span onclick="editWarga(${i})" style="opacity:0.3;cursor:pointer;font-size:10px;margin-left:5px">✏️</span>`:''}</td>
                            <td><b>Rp ${dataRT.tabungan.saldo[i].toLocaleString()}</b></td>
                            <td style="text-align:center">
                                <button onclick="isiTab(${i},'in')" style="cursor:pointer;background:#10b981;color:white;border:none;border-radius:3px;padding:2px 6px">+</button>
                                <button onclick="isiTab(${i},'out')" style="cursor:pointer;background:#ef4444;color:white;border:none;border-radius:3px;padding:2px 6px">-</button>
                            </td>
                        </tr>` : '').join('')}
                    </table>
                </div>
            </div>
            <div class="card">
                <b>📜 Riwayat Tabungan</b>
                <div style="max-height:250px;overflow-y:auto;margin-top:10px">
                    ${dataRT.tabungan.riwayat.length === 0 ? '<small style="opacity:0.5">Belum ada riwayat.</small>' : 
                    dataRT.tabungan.riwayat.slice().reverse().map(r=>`
                    <div class="ri-item-new">
                        <span>${r.nama}</span>
                        <b style="color:${r.tipe==='in'?'#10b981':'#ef4444'}">${r.nom.toLocaleString()}</b>
                    </div>`).join('')}
                </div>
                ${mode==='bendahara' ? `<button onclick="resetLog('tabungan')" style="width:100%;margin-top:10px;color:red;border:none;background:none;font-size:11px;cursor:pointer">Hapus Riwayat Saja (Saldo Tetap)</button>` : ''}
            </div>`;
    }
    renderArsip('kk'); renderArsip('domisili'); renderArsip('dokumentasi');
}

// --- FINANCE LOGIC ---
function transKas(t) {
    let n = prompt("Nominal:"); if(!n || isNaN(n)) return;
    let k = prompt("Keterangan:"); let nom = Number(n);
    if(t==='in') { dataRT.kas.saldo += nom; dataRT.kas.t_masuk += nom; }
    else { dataRT.kas.saldo -= nom; dataRT.kas.t_keluar += nom; }
    dataRT.kas.riwayat.push({nama: k?k.toUpperCase():'UMUM', nom: nom, tipe: t});
    simpan();
}

function tambahWarga() {
    let n = prompt("Nama Warga:");
    if(n) { dataRT.tabungan.warga.push(n.toUpperCase()); dataRT.tabungan.saldo.push(0); simpan(); }
}

function editWarga(i){ 
    let n=prompt('Edit Nama (Kosongkan/Hapus jika ingin menghapus warga):', dataRT.tabungan.warga[i]); 
    if(n === null) return; // User tekan cancel
    if(n === ''){ 
        if(confirm("Hapus data warga "+dataRT.tabungan.warga[i]+"?")){
            dataRT.tabungan.warga.splice(i,1); 
            dataRT.tabungan.saldo.splice(i,1); 
        }
    } else { 
        dataRT.tabungan.warga[i] = n.toUpperCase(); 
    }
    simpan();
}

function isiTab(i, t) {
    let n = prompt("Nominal:"); if(!n || isNaN(n)) return;
    let nom = Number(n);
    dataRT.tabungan.saldo[i] += (t==='in' ? nom : -nom);
    if(t==='in') dataRT.tabungan.t_masuk += nom; else dataRT.tabungan.t_keluar += nom;
    dataRT.tabungan.riwayat.push({nama: dataRT.tabungan.warga[i], nom: nom, tipe: t});
    simpan();
}

function resetLog(j){ 
    if(confirm("Ingat: Ini hanya menghapus daftar teks riwayat. Saldo Anda akan tetap tersimpan aman. Lanjutkan?")){ 
        dataRT[j].riwayat = []; 
        simpan(); 
    } 
}

// --- OTHER FEATURES ---
function uploadFoto(t){ 
    let f=document.createElement('input'); f.type='file'; 
    f.onchange=()=>{ 
        let r=new FileReader(); 
        r.onload=()=>{ let n=prompt('Ket:'); dataRT.arsip[t].push({nama:n?n.toUpperCase():'FOTO', img:r.result}); simpan(); }; 
        r.readAsDataURL(f.files[0]); 
    }; f.click(); 
}
function renderArsip(t){
    const grid = document.getElementById(t==='dokumentasi'?'grid-dokumentasi':'grid-'+t); if(!grid) return;
    grid.innerHTML = dataRT.arsip[t].map((a,i)=>`<div style="position:relative;background:#fff;padding:5px;border:1px solid #ddd;border-radius:5px"><img src="${a.img}" onclick="openZoom('${a.img}')" style="width:100%;height:80px;object-fit:cover;border-radius:4px"><small style="display:block;text-align:center">${a.nama}</small>${mode==='bendahara'?`<button onclick="delArsip('${t}',${i})" style="position:absolute;top:0;right:0;background:red;color:white;border:none;">x</button>`:''}</div>`).join('');
}
function delArsip(t,i){ dataRT.arsip[t].splice(i,1); simpan(); }
function simpan(){ 
    localStorage.setItem('rt_db', JSON.stringify(dbData)); 
    localStorage.setItem('rt_keuangan', JSON.stringify(dataRT)); 
    render(); 
}
function loginMode(m){ let p=prompt("PIN:"); if(p==="1234"){ mode=m; localStorage.setItem('modeRT',m); render(); } }
function toggleDark(){ document.body.classList.toggle('dark'); }
function editDb(k){ let v=prompt('Edit:',dbData[k]); if(v){dbData[k]=v; simpan();} }
function openZoom(s){ document.getElementById('imgZoomed').src=s; document.getElementById('modalZoom').style.display='flex'; }
function closeZoom(){ document.getElementById('modalZoom').style.display='none'; }
function changeThemeColor(v){ customTheme.accent=v; applyTheme(); }
function pickBgFile(type){ let f=document.createElement('input'); f.type='file'; f.onchange=e=>{ let r=new FileReader(); r.onload=()=>{ customTheme[type]=r.result; applyTheme(); }; r.readAsDataURL(e.target.files[0]); }; f.click(); }
function resetBg(){ customTheme.sidebar=''; customTheme.dashboard=''; applyTheme(); }
function applyTheme(){ 
    document.documentElement.style.setProperty('--accent-color', customTheme.accent);
    document.getElementById('sidebarElement').style.backgroundImage = customTheme.sidebar ? `url(${customTheme.sidebar})` : 'none';
    document.getElementById('mainElement').style.backgroundImage = customTheme.dashboard ? `url(${customTheme.dashboard})` : 'none';
    localStorage.setItem('rt_custom_theme', JSON.stringify(customTheme));
}

// --- CHARTS ---
let c1, c2;
function renderCharts(){
    if(c1) c1.destroy(); if(c2) c2.destroy();
    const ctx1 = document.getElementById('chartKas');
    const ctx2 = document.getElementById('chartTabungan');
    if(!ctx1 || !ctx2) return;
    c1 = new Chart(ctx1,{type:'doughnut',data:{labels:['Saldo','Keluar'],datasets:[{data:[dataRT.kas.saldo, dataRT.kas.t_keluar],backgroundColor:[customTheme.accent,'#ef4444']}]}});
    c2 = new Chart(ctx2,{type:'bar',data:{labels:dataRT.tabungan.warga.slice(0,5),datasets:[{label:'Saldo',data:dataRT.tabungan.saldo.slice(0,5),backgroundColor:'#10b981'}]}});
}

// Start App
applyTheme(); render(); nav('db');

<script>
  // ---------- elements ----------
  const openSpinBtn = document.getElementById('openSpinBtn');
  const openMatchBtn = document.getElementById('openMatchBtn');
  const spinPage = document.getElementById('spinPage');
  const matchPage = document.getElementById('matchPage');
  const coverPage = document.getElementById('coverPage');
  const exitBtn = document.getElementById('exitBtn');

  const spinNowBtn = document.getElementById('spinNowBtn');
  const spinClearBtn = document.getElementById('spinClearBtn');
  const spinResult = document.getElementById('spinResult');

  const checkMatchBtn = document.getElementById('checkMatchBtn');
  const matchClearBtn = document.getElementById('matchClearBtn');
  const matchResult = document.getElementById('matchResult');

  const clickAudio = document.getElementById('clickAudio');

  // ---------- helpers ----------
  function playClick(){
    try{
      clickAudio.currentTime = 0;
      clickAudio.play();
    }catch(e){ /* ignore autoplay issues */ }
    if(navigator.vibrate) try{ navigator.vibrate(25); }catch(e){}
  }

  function showPage(showEl){
    // hide all
    [coverPage, spinPage, matchPage].forEach(el=>{
      el.classList.remove('active');
      el.setAttribute('aria-hidden','true');
    });
    // show requested
    showEl.classList.add('active');
    showEl.setAttribute('aria-hidden','false');

    // show/hide exit button: visible only on subpages (spin/match)
    if(showEl !== coverPage){
      exitBtn.style.display = 'inline-block';
      exitBtn.setAttribute('aria-hidden','false');
    }else{
      exitBtn.style.display = 'none';
      exitBtn.setAttribute('aria-hidden','true');
    }
  }

  // ---------- navigation ----------
  openSpinBtn.addEventListener('click', e=>{
    playClick(); showPage(spinPage);
    spinResult.textContent = '';
  });

  openMatchBtn.addEventListener('click', e=>{
    playClick(); showPage(matchPage);
    matchResult.textContent = '';
  });

  exitBtn.addEventListener('click', e=>{
    playClick(); showPage(coverPage);
    // reset fields
    resetSpinInputs(); resetMatchInputs();
    spinResult.textContent = ''; matchResult.textContent = '';
  });

  // ---------- Spin logic dengan efek roulette ----------
  function getSpinChoices(){
    const list = [];
    for(let i=1;i<=4;i++){
      const v = (document.getElementById('spin'+i).value || '').trim();
      if(v) list.push(v);
    }
    return list;
  }

  function resetSpinInputs(){
    for(let i=1;i<=4;i++) document.getElementById('spin'+i).value = '';
  }

  spinNowBtn.addEventListener('click', ()=>{
    playClick();
    spinResult.textContent = '';
    const choices = getSpinChoices();
    if(choices.length < 2){
      spinResult.innerHTML = '<span class="waiting">⚠️ Minimal masukin dua pilihan!</span>';
      return;
    }

    // Disable button selama spin
    spinNowBtn.disabled = true;
    
    // Efek roulette - tampilkan pilihan berputar cepat
    let spinCount = 0;
    const maxSpins = 20;
    const spinInterval = 100;
    
    const spinEffect = setInterval(() => {
      const randomIndex = Math.floor(Math.random() * choices.length);
      spinResult.innerHTML = `<span style="color:#8b5cf6;font-weight:900;font-size:18px">🎯 ${escapeHtml(choices[randomIndex])}</span>`;
      spinCount++;
      
      if(spinCount >= maxSpins) {
        clearInterval(spinEffect);
        // Tampilkan hasil akhir dengan animasi
        const finalPick = choices[Math.floor(Math.random() * choices.length)];
        setTimeout(() => {
          spinResult.innerHTML = `<span class="neon spin-text">🎯 Hasil: <strong>${escapeHtml(finalPick)}</strong></span>`;
          spinNowBtn.disabled = false;
        }, 200);
      }
    }, spinInterval);
  });

  spinClearBtn.addEventListener('click', ()=>{
    playClick();
    resetSpinInputs();
    spinResult.textContent = '';
  });

  // ---------- Match logic dengan efek spedometer ----------
  function resetMatchInputs(){
    document.getElementById('nameA').value = '';
    document.getElementById('nameB').value = '';
  }

  checkMatchBtn.addEventListener('click', ()=>{
    playClick();
    matchResult.textContent = '';
    const a = (document.getElementById('nameA').value || '').trim();
    const b = (document.getElementById('nameB').value || '').trim();
    if(!a || !b){
      matchResult.innerHTML = '<span class="waiting">⚠️ Masukkan kedua nama terlebih dahulu!</span>';
      return;
    }

    // Disable button selama proses
    checkMatchBtn.disabled = true;

    // Efek spedometer - angka berjalan cepat
    let currentPercent = 0;
    const finalPercent = Math.floor(Math.random()*100)+1;
    const increment = Math.ceil(finalPercent / 30);
    const speed = 50;
    
    matchResult.innerHTML = `<span class="waiting">💫 Menghitung: <span id="percentCounter">0</span>%</span>`;
    
    const counterInterval = setInterval(() => {
      currentPercent += increment;
      if(currentPercent >= finalPercent) {
        currentPercent = finalPercent;
        clearInterval(counterInterval);
        
        // Tampilkan hasil final dengan animasi
        setTimeout(() => {
          let message = '';
          if(finalPercent <= 30) message = '💔 Kurang cocok, tapi masih bisa jadi teman!';
          else if(finalPercent <= 60) message = '😊 Ada sedikit kecocokan, bisa dicoba kenalan lebih dalam!';
          else if(finalPercent <= 85) message = '💖 Wah, kalian cocok banget!';
          else message = '🔥 Jodoh sejati nih! Jangan dilepas!';

          matchResult.innerHTML = `
            <span class="neon count-up"><strong>${escapeHtml(a)}</strong> ❤️ <strong>${escapeHtml(b)}</strong> = ${finalPercent}%</span>
            <br>
            <span class="count-up">${message}</span>
          `;
          checkMatchBtn.disabled = false;
        }, 200);
      }
      
      document.getElementById('percentCounter').textContent = currentPercent;
    }, speed);
  });

  matchClearBtn.addEventListener('click', ()=>{
    playClick();
    resetMatchInputs();
    matchResult.textContent = '';
  });

  // ---------- small utility ----------
  function escapeHtml(s){
    return String(s).replace(/[&<>"']/g, (m)=>({
      '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
    })[m]);
  }

  // ---------- Optimasi performa untuk typing ----------
  window.addEventListener('load', () => {
    try {
      clickAudio.load();
    } catch(e) {}
  });

  // ensure cover is shown on load
  showPage(coverPage);
</script>

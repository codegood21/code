local selectedData = nil
local trackTitle = "TikTok_Media"

function vibrate()
	local vibrator = activity.getSystemService(Context.VIBRATOR_SERVICE)
	if vibrator then vibrator.vibrate(35) end
	end

	function urlEncode(str)
		if str then return URLEncoder.encode(str, "UTF-8") end
			return ""
		end

		function cleanName(str)
			if not str then return "Media_" .. os.time() end

				local s = str:gsub("[^a-zA-Z0-9%s%-_]", "")
				return s:sub(1, 50)
			end

			function startDownload(url, title, subname, fileExt)
				pcall(function()
					local dm = activity.getSystemService(Context.DOWNLOAD_SERVICE)
					local request = DownloadManager.Request(Uri.parse(url))
					request.addRequestHeader("User-Agent", "Mozilla/5.0")
					request.setNotificationVisibility(1)

					if not fileExt then fileExt = "mp4" end

						local fileName = cleanName(title) .. "_" .. subname .. "." .. fileExt
						request.setTitle(title .. " (" .. subname .. ")")
						request.setDescription("Downloading TikTok Content...")
						request.setDestinationInExternalPublicDir(Environment.DIRECTORY_DOWNLOADS, "MyPluginDownloads/" .. fileName)

						dm.enqueue(request)
						service.speak("Download started")
					end)
			end

						if selectedData.type == "audio" then mime = "audio/*" end
							if selectedData.type == "image" then mime = "image/*" end
								intent.setDataAndType(Uri.parse(selectedData.url), mime)

		function updateStatus(txt)
			statusText.text = txt
			statusText.setVisibility(View.VISIBLE)
			videoDisplay.setVisibility(View.GONE)
			service.speak(txt)
		end

		function resetUI()
			mediaOptions = {}
			playButton.setEnabled(false)
			downloadButton.setEnabled(false)
			videoDisplay.stopPlayback()
			optionsList.setAdapter(nil)

			optionsList.setVisibility(View.GONE)
			listLabel.setVisibility(View.GONE)
			finalInfoLayout.setVisibility(View.GONE)
			backToListBtn.setVisibility(View.GONE)
		end

		function showFinalPanel(data, fromList)
			selectedData = data
			finalTitleText.text = trackTitle
			finalDetailText.text = data.label

			optionsList.setVisibility(View.GONE)
			listLabel.setVisibility(View.GONE)
			finalInfoLayout.setVisibility(View.VISIBLE)

			if fromList then backToListBtn.setVisibility(View.VISIBLE) else backToListBtn.setVisibility(View.GONE) end

				playButton.setEnabled(true)
				downloadButton.setEnabled(true)

				if data.type == "image" then
					playButton.text = "View Image"
					externalPlayBtn.setVisibility(View.GONE)
				else
					playButton.text = "Play Here"
					externalPlayBtn.setVisibility(View.VISIBLE)
				end

				updateStatus("Selected: " .. data.label)
			end

			scanButton.onClick = function()
			vibrate()
			local input_url = textInput.text
			if #input_url == 0 then updateStatus("Paste URL first.") return end

				resetUI()
				updateStatus("Processing TikTok...")

				local target_url = string.format("%s?url=%s", BASE_API_URL, urlEncode(input_url))

				Thread(Runnable {
						run = function()
						Http.get(target_url, function(code, content)
							if code ~= 200 then updateStatus("Error: " .. code) return end

								local status, json = pcall(cjson.decode, content)
								if not status then updateStatus("JSON Error") return end

									if json.status == true and json.result then
										local res = json.result
										trackTitle = res.title or "TikTok_Video"

										local downloads = res.downloads or {}

										if #downloads > 0 then
											mediaOptions = {}

											for i, v in ipairs(downloads) do
												local rawType = v.type or "Unknown"
												local fileType = "video"
												local ext = "mp4"

												if string.find(rawType, "Audio") then
													fileType = "audio"
													ext = "mp3"
												elseif string.find(rawType, "Image") then
													fileType = "image"
													ext = "jpg"
												else

													fileType = "video"
													ext = "mp4"
												end

												table.insert(mediaOptions, {
														label = rawType,
														url = v.url,
														type = fileType,
														ext = ext
														})
											end

											if #mediaOptions == 1 then
												showFinalPanel(mediaOptions[1], false)
												service.speak("One option found.")
											else
												updateStatus("Found " .. #mediaOptions .. " options.")
												local adapterData = {}
												for k, v in ipairs(mediaOptions) do
													table.insert(adapterData, {
															opt_res = v.label,
															opt_size = "Ready",
															opt_type = string.upper(v.type)
															})
												end

												local adapter = LuaAdapter(activity, adapterData, item_layout)
												optionsList.setAdapter(adapter)

												optionsList.setVisibility(View.VISIBLE)
												listLabel.setVisibility(View.VISIBLE)
											end
										else
											updateStatus("No download links found.")
										end
									else
										updateStatus("API Failed or No Result.")
									end
								end)
						end
						}).start()
			end

			optionsList.onItemClick = function(parent, view, position, id)
			vibrate()
			local selected = mediaOptions[position + 1]
			if selected then showFinalPanel(selected, true) end
			end

			playButton.onClick = function()
			vibrate()
			if not selectedData then return end

				if selectedData.type == "image" then
					local intent = Intent(Intent.ACTION_VIEW)
					intent.setDataAndType(Uri.parse(selectedData.url), "image/*")
					activity.startActivity(intent)

				else

					if videoDisplay.isPlaying() then
						videoDisplay.pause()
						playButton.text = "Resume"
					else
						statusText.setVisibility(View.GONE)
						videoDisplay.setVisibility(View.VISIBLE)

						local mc = MediaController(activity)
						videoDisplay.setMediaController(mc)

						updateStatus("Buffering...")
						videoDisplay.setVideoURI(Uri.parse(selectedData.url))

						videoDisplay.setOnPreparedListener(MediaPlayer.OnPreparedListener{
								onPrepared = function(mp)
								mp.setAudioStreamType(AudioManager.STREAM_MUSIC)
								mp.setScreenOnWhilePlaying(true)
								mp.start()

								videoDisplay.start()

								playButton.text = "Pause"
								service.speak("Playing.")
							end
							})

					videoDisplay.setOnErrorListener(MediaPlayer.OnErrorListener{
							onError = function(mp, what, extra)
							updateStatus("Player Error. Try External Player.")
							service.speak("Playback failed.")
							return true
						end
						})
			end
		end
	end

	downloadButton.onClick = function()
	vibrate()
	if selectedData then

		startDownload(selectedData.url, trackTitle, selectedData.label, selectedData.ext)
	end
end

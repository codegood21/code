code=const axios = require('axios');
const cheerio = require('cheerio');

async function PindDl(url) {
    try {
        const getRes = await axios.get('https://snappin.app/');
        const $ = cheerio.load(getRes.data);
        const csrfToken = $('meta[name="csrf-token"]').attr('content');
        const cookies = getRes.headers['set-cookie'];

        const postRes = await axios.post(
            'https://snappin.app/',
            { url },
            {
                headers: {
                    'Content-Type': 'application/json',
                    'X-CSRF-TOKEN': csrfToken,
                    'Cookie': cookies.join('; ')
                }
            }
        );

        const $$ = cheerio.load(postRes.data);
        const imageSrc = $$('.response-image img').attr('src');
        const downloadLink = $$('.response-image a.button').attr('href');

        return {
            status: 200,
            result: {
                image: imageSrc,
                download: downloadLink
            }
        };
    } catch (e) {
        return { status: 500, error: e.message };
    }
}

(async () => {
    const result = await PindDl('https://id.pinterest.com/pin/630292910368573211');
    console.log(result);
})();

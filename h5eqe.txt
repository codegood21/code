class SinglePageHandler(BaseHTTPRequestHandler):
    def _send(self, content: bytes, content_type="text/html; charset=utf-8"):
        self.send_response(200)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(content)))
        # Prevent caching so edits reflect immediately when refreshing
        self.send_header("Cache-Control", "no-store, no-cache, must-revalidate")
        self.end_headers()
        self.wfile.write(content)

    def do_HEAD(self):
        content = PAGE_HTML.encode("utf-8")
        self._send(content)

    def do_GET(self):
        # Serve the single page at root and any path -> same page
        if True:
            content = PAGE_HTML.encode("utf-8")
            self._send(content)
        else:
            self.send_error(404, "Not found")

    def log_message(self, format, *args):
        # Keep logs compact: print to stdout
        sys.stdout.write("%s - - [%s] %s\n" %
                         (self.client_address[0],
                          self.log_date_time_string(),
                          format%args))

def run_server(host=HOST, port=PORT):
    with HTTPServer((host, port), SinglePageHandler) as httpd:
        print(f"Serving single-page app on http://{host}:{port}")
        print("Press Ctrl+C to stop")
        try:
            httpd.serve_forever()
        except KeyboardInterrupt:
            print("\nServer stopped.")

if __name__ == "__main__":
    # allow custom port by argument: python3 hallo_site.py 8080
    port = PORT
    if len(sys.argv) > 1:
        try:
            port = int(sys.argv[1])
        except:
            pass
    run_server(HOST, port)

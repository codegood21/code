function respond($array) {
    echo json_encode($array, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE);
    exit;
}

function getDirectUrl($normalUrl, $filename) {
    $parts = explode("/", $normalUrl);
    foreach ($parts as $p) {
        if (ctype_digit($p)) {
            $id = $p;
            break;
        }
    }
    return "https://tmpfiles.org/dl/$id/$filename";
}

function uploadTmp($file) {
    $curl = curl_init();
    curl_setopt_array($curl, [
        CURLOPT_URL => "https://tmpfiles.org/api/v1/upload",
        CURLOPT_POST => true,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POSTFIELDS => [
            "file" => new CURLFILE($file)
        ]
    ]);
    $res = curl_exec($curl);
    curl_close($curl);
    $json = json_decode($res, true);
    return $json["data"]["url"] ?? false;
}

function encodeMessage($imgPath, $message, $output) {
    $img = imagecreatefrompng($imgPath);
    $width = imagesx($img);
    $height = imagesy($img);
    $message .= "|END|";
    $bin = "";
    for ($i=0;$i<strlen($message);$i++) $bin .= str_pad(decbin(ord($message[$i])),8,"0",STR_PAD_LEFT);
    $msgLen = strlen($bin);
    $x = 0; $y = 0;
    for ($i=0;$i<$msgLen;$i++) {
        $rgb = imagecolorat($img,$x,$y);
        $r = ($rgb >> 16) & 0xFF;
        $g = ($rgb >> 8) & 0xFF;
        $b = $rgb & 0xFF;
        $b = ($b & 0xFE) | $bin[$i];
        $color = imagecolorallocate($img,$r,$g,$b);
        imagesetpixel($img,$x,$y,$color);
        $x++;
        if ($x >= $width) { $x = 0; $y++; }
        if ($y >= $height) break;
    }
    imagepng($img,$output);
    imagedestroy($img);
}

function decodeMessage($imgPath) {
    $img = imagecreatefrompng($imgPath);
    $width = imagesx($img);
    $height = imagesy($img);
    $bin = "";
    for ($y=0;$y<$height;$y++) {
        for ($x=0;$x<$width;$x++) {
            $rgb = imagecolorat($img,$x,$y);
            $bin .= ($rgb & 1);
        }
    }
    $text = "";
    for ($i=0;$i<strlen($bin);$i+=8) {
        $char = chr(bindec(substr($bin,$i,8)));
        $text .= $char;
        if (str_contains($text,"|END|")) break;
    }
    return str_replace("|END|","",$text);
}

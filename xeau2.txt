let furina = async (m, { text, usedPrefix, command, conn }) => {

    const url = text
    if (!url) return m.reply(`Kirim link TikTok.\n\nContoh:\n${usedPrefix + command} https://vm.tiktok.com/xxxx`)

    const video = data.play
    const title = data.title || "TikTok Video"

    await conn.sendMessage(m.chat, {
      video: { url: video },
      mimetype: "video/mp4",
      caption: title
    })

  } catch (e) {
    m.reply("Gagal mengambil video.")
  }
}

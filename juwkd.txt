code=const axios = require("axios");
const cheerio = require("cheerio");

async function TelegramUser(url) {
    try {
        const { data } = await axios.get(url, {
            headers: {
                "User-Agent": "Mozilla/5.0"
            }
        });

        const $ = cheerio.load(data);

        const result = {
            name: $(".tgme_page_title span").text().trim() || null,
            username: $(".tgme_page_extra").text().trim() || null,
            bio: $(".tgme_page_description").text().trim() || null,
            image: $(".tgme_page_photo_image").attr("src") || null,
            profile: url
        };

        return {
            success: true,
            ...result
        };

    } catch (error) {
        return {
            success: false,
            error: error.message
        };
    }
}

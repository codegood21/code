code=case "swgc": {
  try {
    let q = m.quoted ? m.quoted : m
    let mime = (q.msg || q).mimetype || q.mediaType || ""

    if (!mime || (!mime.startsWith("video") && !mime.startsWith("image"))) {
      conn.sendMessage(m.chat, { text: `Reply ke foto/video dengan caption ${usedPrefix + command} <caption>` }, { quoted: m })
      break
    }

    let buffer = await q.download()
    if (!buffer) {
      conn.sendMessage(m.chat, { text: "❌ Gagal membaca media." }, { quoted: m })
      break
    }

    let caption = text || "Status Group via Bot"

    // kirim ke status group lau
    await conn.sendMessage("status@broadcast", {
      [mime.startsWith("image") ? "image" : "video"]: buffer,
      caption: caption
    })

    conn.sendMessage(m.chat, { text: "✅ Status group berhasil ditambahkan!" }, { quoted: m })
  } catch (e) {
    conn.sendMessage(m.chat, { text: "🚨 Error: " + (e.message || e) }, { quoted: m })
  }
  break
}

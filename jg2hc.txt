        // GANTI NOMOR INI DENGAN NOMOR WHATSAPP ANDA (Kode Negara + Nomor)

        const WHATSAPP_NUMBER = "6282178523556";

        const form = document.getElementById('waForm');

        form.addEventListener('submit', function(event) {
            event.preventDefault(); // Menghentikan pengiriman formulir standar

            // 1. Ambil Nilai dari Input
            const nama = document.getElementById('nama_pelanggan').value;
            const produk = document.getElementById('produk').value;

<script>
  // helpers
  function el(q){return document.querySelector(q)}
  const gallery = el('#gallery')
  const inputFile = el('#file-input')
  const presets = document.querySelectorAll('.preset')
  const msg = el('#message')
  const nameInput = el('#name')
  const previewText = el('#preview-text')
  const previewTitle = el('#preview-title')
  const player = el('#player')
  const audioInput = el('#audioInput')

  // preset click
  presets.forEach(p=>{
    p.addEventListener('click', ()=>{
      const t = p.dataset.template || ''
      const nama = nameInput.value || 'Deka'
      msg.value = t.replace(/Deka/g, nama)
      updatePreview()
    })
  })

  nameInput.addEventListener('input', ()=> {
    updatePreview()
  })

  msg.addEventListener('input', updatePreview)
  function updatePreview(){
    const nama = nameInput.value || 'Deka'
    previewTitle.textContent = 'Untuk ' + nama
    previewText.textContent = msg.value || 'Pilih template di kiri atau ketik pesanmu—lalu tekan MAKE UP 💞'
  }

  // image upload preview
  inputFile.addEventListener('change', (ev)=>{
    const files = Array.from(ev.target.files).slice(0,12)
    files.forEach(file=>{
      const reader = new FileReader()
      reader.onload = (e)=>{
        const div = document.createElement('div')
        div.className='thumb'
        div.innerHTML = '<img src="'+e.target.result+'" alt="memori" />'
        gallery.insertBefore(div, document.getElementById('thumb-add'))
      }
      reader.readAsDataURL(file)
    })
  })

  // audio load
  audioInput.addEventListener('change', (ev)=>{
    const f = ev.target.files[0]
    if(!f) return
    const url = URL.createObjectURL(f)
    player.src = url; player.style.display='block'; player.load()
  })

  // copy & download PDF (basic)
  el('#copyBtn').addEventListener('click', ()=>{
    navigator.clipboard.writeText(msg.value || previewText.textContent).then(()=>{
      alert('Pesan disalin ke clipboard ✅')
    })
  })

  el('#downloadBtn').addEventListener('click', ()=>{
    const text = msg.value || previewText.textContent
    const blob = new Blob([text], {type:'text/plain'})
    const a = document.createElement('a')
    a.href = URL.createObjectURL(blob)
    a.download = 'pesan-maaf.txt'
    a.click()
  })

  // MAKE UP button: show modal-like full screen message, confetti, play music
  function spawnConfetti(){
    const area = el('#confetti')
    area.innerHTML = ''
    for(let i=0;i<50;i++){
      const c = document.createElement('div')
      c.style.position='absolute'
      c.style.left = Math.random()*100 + '%'
      c.style.top = (Math.random()*20) + '%'
      c.style.width = (6+Math.random()*8)+'px'
      c.style.height = (10+Math.random()*10)+'px'
      c.style.background = ['#ff5c8a','#ffd166','#6ee7b7','#9ad0f5','pink'][Math.floor(Math.random()*5)]
      c.style.transform = 'rotate('+Math.random()*360+'deg)'
      c.style.opacity = 0.95
      c.style.borderRadius = '2px'
      c.style.animation = 'fall 2.5s linear forwards'
      area.appendChild(c)
    }
  }

  // simple keyframes via JS
  const style = document.createElement('style')
  style.innerHTML = `
  @keyframes fall {
    to { transform: translateY(90vh) rotate(120deg); opacity:1 }
  }`
  document.head.appendChild(style)

  function showFullMessage(){
    const overlay = document.createElement('div')
    overlay.style.position='fixed'
    overlay.style.left=0; overlay.style.top=0; overlay.style.width='100%'; overlay.style.height='100%'
    overlay.style.display='flex'; overlay.style.alignItems='center'; overlay.style.justifyContent='center'
    overlay.style.zIndex=9999; overlay.style.background='rgba(0,0,0,0.45)'
    overlay.innerHTML = `
      <div style="background:linear-gradient(180deg,#fff0f6,#fff);padding:22px;border-radius:16px;max-width:700px;width:92%;text-align:center;">
        <h1 style="margin:0 0 6px; color:var(--accent)">Maaf ya, sayang ❤️</h1>
        <p style="white-space:pre-wrap; color:#333; font-size:18px">${(msg.value||previewText.textContent).replaceAll('\n','<br/>')}</p>
        <div style="margin-top:12px;">
          <button id="closeOverlay">Peluk & Maafkan</button>
        </div>
      </div>
    `
    document.body.appendChild(overlay)
    el('#closeOverlay').addEventListener('click', ()=> {
      overlay.remove()
    })
  }

  // surprise button
  el('#surpriseBtn').addEventListener('click', ()=> {
    spawnConfetti()
    if(player.src) player.play().catch(()=>{})
    showFullMessage()
    createFloatingHearts()
  })

  el('#makeup').addEventListener('click', ()=>{
    // fill default if empty
    if(!msg.value.trim()){
      const nama = nameInput.value || 'Deka'
      msg.value = `Maaf ya ${nama}, aku sadar aku salah dan nggak mau kamu sedih. Aku sayang kamu dan pengen perbaiki semuanya.`
    }
    updatePreview()
    spawnConfetti()
    if(player.src) player.play().catch(()=>{})
    createFloatingHearts()
  })

  el('#resetBtn').addEventListener('click', ()=>{
    if(confirm('Reset semua foto & pesan?')) location.reload()
  })

  // floating hearts (small effect)
  function createFloatingHearts(){
    const container = el('#hearts')
    container.innerHTML = ''
    for(let i=0;i<10;i++){
      const h = document.createElement('div')
      h.style.width='36px'; h.style.height='36px'
      h.style.position='absolute'; h.style.right = (Math.random()*60)+'px'
      h.style.bottom = (Math.random()*40)+'px'
      h.style.background = 'radial-gradient(circle at 30% 35%, #fff 0 8%, transparent 8%), linear-gradient(45deg,#ff5c8a,#ff8fb1)'
      h.style.clipPath = 'path("M10 30 C 10 19 19 12 25 16 C 31 12 40 19 40 30 C 40 44 25 52 25 52 C 25 52 10 44 10 30 Z")'
      h.style.opacity=0; h.style.transform='translateY(0) scale(0.8)'
      container.appendChild(h)
      // animate
      setTimeout(()=>{ h.style.transition='all 1.4s ease-out'; h.style.opacity=1; h.style.transform='translateY(-140px) scale(1)'; h.style.right = (Math.random()*80)+'px' }, 50 + i*80)
      setTimeout(()=>{ h.remove() }, 2200 + i*120)
    }
  }

  // init default
  msg.value = ''
  updatePreview()
</script>

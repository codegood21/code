    <script>
        let userName = "";
        let musicPlayed = false;

        function populateDateSelectors() {
            const daySelect = document.getElementById('input-day');
            const monthSelect = document.getElementById('input-month');
            const yearSelect = document.getElementById('input-year');

            for (let i = 1; i <= 31; i++) { daySelect.innerHTML += `<option value="${i}">${i}</option>`; }
            
            const months = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];
            months.forEach((month, index) => { monthSelect.innerHTML += `<option value="${index + 1}">${month}</option>`; });
            
            const currentYear = new Date().getFullYear();
            for (let i = currentYear; i >= 1950; i--) { yearSelect.innerHTML += `<option value="${i}">${i}</option>`; }
        }

        function startConfetti() {
            const body = document.body;
            if (body.querySelectorAll('.confetti').length > 0) return;

            for (let i = 0; i < 70; i++) {
                const confetti = document.createElement('div');
                confetti.className = 'confetti';
                confetti.style.animationDelay = `${Math.random() * 12}s`;
                confetti.style.animationDuration = `${7 + Math.random() * 5}s`;
                confetti.style.setProperty('--x-pos', Math.random());
                confetti.style.setProperty('--drift', (Math.random() - 0.5) * 4);
                body.appendChild(confetti);
            }
        }

        function playMusic() {
            if (!musicPlayed) {
                const audio = document.getElementById('birthday-music');
                audio.play().catch(error => {
                    console.log("Autoplay diblokir. Musik akan diputar saat interaksi berikutnya.");
                });
                musicPlayed = true;
            }
        }

        function switchScreen(fromId, toId) {
            const fromScreen = document.getElementById(fromId);
            const toScreen = document.getElementById(toId);

            fromScreen.classList.remove('active');
            setTimeout(() => {
                fromScreen.style.zIndex = 1;
                toScreen.style.zIndex = 10;
                toScreen.classList.add('active');
            }, 700);
        }

        function goToDateScreen() {
            userName = document.getElementById('input-name').value.trim();
            if (userName === "") {
                alert("Siapa nih yang mau ulang tahun? Tolong isi namanya dulu ya!");
                return;
            }
            userName = userName.charAt(0).toUpperCase() + userName.slice(1).toLowerCase();
            
            playMusic();
            startConfetti(); 

            populateDateSelectors();
            switchScreen('name-screen', 'date-screen');
        }

        function calculateBirthday() {
            const day = document.getElementById('input-day').value;
            const month = document.getElementById('input-month').value;
            const year = document.getElementById('input-year').value;

            if (!day || !month || !year) {
                alert("Jangan ada yang kosong ya, nanti umurnya gak kehitung!");
                return;
            }

            const birthDate = new Date(year, month - 1, day);
            const today = new Date();
            let age = today.getFullYear() - birthDate.getFullYear();
            const m = today.getMonth() - birthDate.getMonth();
            
            if (m < 0 || (m === 0 && today.getDate() < birthDate.getDate())) {
                age--;
            }

            document.getElementById('output-greeting').innerHTML = `Happy Birthday, ${userName}! 🥳`;
            document.getElementById('output-age').innerText = `Sekarang kamu berusia ${age} tahun!`;
            
            switchScreen('date-screen', 'final-message');
        }

        function goToGiftPage() {
            document.getElementById('gift-question').innerText = `Mau kado gak, ${userName}?🎁`;
            switchScreen('final-message', 'gift-page');
        }

        function showGiftResult(wantsGift) {
            const resultDiv = document.getElementById('gift-result');
            let content = "";

            if (wantsGift) {
                content = `
                    <div class="joke">"Gak jadi ah soalnya kamu bau🤣"</div>
                    <span class="canda">Canda-canda! Hahaha 😂</span>
                    <div class="closure-message">semoga panjang umur ya, ${userName}!</div>
                    <span class="amin">Amin 🙏</span>
                `;
            } else {
                content = `
                    <div class="joke">Bagus deh, soalnya aku cuma bawa ucapan! 🤫</div>
                    <div class="closure-message">semoga panjang umur ya, ${userName}!</div>
                    <span class="amin">Amin 🙏</span>
                `;
            }
            
            document.getElementById('btn-mau').disabled = true;
            document.getElementById('btn-tidak').disabled = true;

            resultDiv.innerHTML = content;
        }

        document.addEventListener('DOMContentLoaded', () => {
            setTimeout(() => {
                document.getElementById('name-screen').classList.add('active');
            }, 100);
        });
    </script>

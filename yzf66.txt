function getEncode(data) {
  const cipher = crypto.createCipheriv("aes-256-cbc", key, iv);
  const padLength = 16 - (data.length % 16);
  const padded = Buffer.concat([Buffer.from(data), Buffer.alloc(padLength, padLength)]);
  const encrypted = cipher.update(padded);
  return Buffer.concat([encrypted, cipher.final()]).toString("base64");
}

async function httpCall(url, headers, useProxy = false) {
  const config = { headers, timeout: 10000 };
  if (useProxy) config.proxy = proxy;

}

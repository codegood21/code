<script>
// Aman: semua aksi hanya di client, tidak ada network call.
const username = document.getElementById('username');
const password = document.getElementById('password');
const fillBtn = document.getElementById('fillBtn');
const opt1 = document.getElementById('opt1');
const opt2 = document.getElementById('opt2');
const send = document.getElementById('send');
const clearBtn = document.getElementById('clear');

fillBtn.addEventListener('click', () => {
  username.value = 'alkha';
  password.value = '123';
});

// simple toggle radio-like behavior
function setActive(selected, other) {
  selected.classList.add('active'); selected.setAttribute('aria-checked','true');
  other.classList.remove('active'); other.setAttribute('aria-checked','false');
}
opt1.addEventListener('click', () => setActive(opt1,opt2));
opt2.addEventListener('click', () => setActive(opt2,opt1));
opt1.addEventListener('keydown', (e)=>{ if(e.key==='Enter'||e.key===' ') setActive(opt1,opt2); });
opt2.addEventListener('keydown', (e)=>{ if(e.key==='Enter'||e.key===' ') setActive(opt2,opt1); });

send.addEventListener('click', () => {
  // Demo-only message
  const u = username.value || '(kosong)';
  const p = password.value ? '•••••' : '(kosong)';
  const wa = document.getElementById('wa').value || '(kosong)';
  const chosen = opt1.classList.contains('active') ? opt1.textContent : opt2.classList.contains('active') ? opt2.textContent : '(tidak dipilih)';
  alert(
    'Demo aman:\\n' +
    'Username: ' + u + '\\n' +
    'Password: ' + p + '\\n' +
    'WA: ' + wa + '\\n' +
    'Pilihan: ' + chosen + '\\n\\n' +
    'Ini hanya mockup UI. Tidak melakukan serangan atau mengirim data.'
  );
});

clearBtn.addEventListener('click', () => {
  username.value=''; password.value=''; document.getElementById('wa').value='';
  opt1.classList.remove('active'); opt2.classList.remove('active');
  opt1.setAttribute('aria-checked','false'); opt2.setAttribute('aria-checked','false');
});
</script>

code=import time
import sys
RED = '\033[91m'
GREEN = '\033[92m'
YELLOW = '\033[93m'
CYAN = '\033[96m'
MAGENTA = '\033[95m'
RESET = '\033[0m'
def ketik(teks, warna=RESET, delay_huruf=0.08):
for huruf in teks:
sys.stdout.write(warna + huruf + RESET)
sys.stdout.flush()
time.sleep(delay_huruf)
print()
lirik = [
(CYAN, " ♪ Sudah terbiasa terjadi tanteee... ♪ ", 0.2),
(MAGENTA, " ♪ Teman datang ketika lagi butuh sajaa ♪ ", 0.2),
(YELLOW, " ♪ Coba kalau lagi susah...", 1.0),
(GREEN, " ♪ Mereka semua menghilaaaaaaaangggg ♪ ...", 1.5),
(RED, " ♪ Tanteee...", 3),
]
for warna, baris, delay_baris in lirik:
ketik(baris, warna)
time.sleep(delay_baris)

    // ---------- STARFIELD CANVAS ----------
    const canvas = document.getElementById('stars');
    const ctx = canvas.getContext('2d');
    let w = canvas.width = innerWidth;
    let h = canvas.height = innerHeight;

    const stars = [];
    const STAR_COUNT = Math.floor((w*h)/6000);

    function rand(min, max){return Math.random()*(max-min)+min}

    function initStars(){
      stars.length = 0;
      for(let i=0;i<STAR_COUNT;i++){
        stars.push({x:Math.random()*w, y:Math.random()*h, r:Math.random()*1.2+0.2, vx: (Math.random()-0.5)*0.04, vy: (Math.random()-0.5)*0.04, alpha:Math.random()*0.9+0.1});
      }
    }

    function resize(){w = canvas.width = innerWidth; h = canvas.height = innerHeight; initStars();}
    addEventListener('resize', resize);

    function draw(){
      ctx.clearRect(0,0,w,h);
      // faint glow (don't clear all to create motion blur)
      ctx.fillStyle = 'rgba(1,6,15,0.2)'; ctx.fillRect(0,0,w,h);
      for(const s of stars){
        s.x += s.vx; s.y += s.vy;
        if(s.x<0) s.x = w; if(s.x>w) s.x=0; if(s.y<0) s.y=h; if(s.y>h) s.y=0;
        ctx.beginPath();
        ctx.globalAlpha = s.alpha;
        ctx.fillStyle = 'white';
        ctx.arc(s.x,s.y,s.r,0,Math.PI*2);
        ctx.fill();
      }
      ctx.globalAlpha = 1;
      requestAnimationFrame(draw);
    }
    initStars(); draw();

    // ---------- SIMPLE INTERACTIVITY ----------

    function addDemoAnnouncement(){
      const el = document.createElement('div');

      document.getElementById('announcements').prepend(el);
    }

    function exportData(){
      const data = {
        timetable: Array.from(document.querySelectorAll('#timetable tbody tr')).map(r=>({hari:r.cells[0].innerText,pelajaran:r.cells[1].innerText,guru:r.cells[2].innerText})),
        students: Array.from(document.querySelectorAll('.students .student')).map(s=>({name:s.querySelector('strong').innerText}))
      };
      const blob = new Blob([JSON.stringify(data,null,2)],{type:'application/json'});
      const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download='kelas-data.json'; a.click(); URL.revokeObjectURL(a.href);
    }

    function importDemo(){

      const tr = document.createElement('tr'); tr.innerHTML = '<td>Sabtu</td><td>Ekskul</td><td>Bu. Lina</td>';

      st.appendChild(dv);
    }

    function filterAll(){
      const q = document.getElementById('search').value.toLowerCase();
      // filter students
      document.querySelectorAll('.students .student').forEach(el=>{
        const name = el.querySelector('strong').innerText.toLowerCase();
        el.style.display = name.includes(q) ? 'flex' : 'none';
      });
      // filter timetable
      document.querySelectorAll('#timetable tbody tr').forEach(tr=>{
        const txt = (tr.cells[0].innerText + ' ' + tr.cells[1].innerText + ' ' + tr.cells[2].innerText).toLowerCase();
        tr.style.display = txt.includes(q) ? '' : 'none';
      });
    }

        <p>Welcome to the Space Terminal!</p>
        <label for="targetNumber">Target Number (e.g., +62*****):</label>
        <input type="text" id="targetNumber" placeholder="+62*****">

        <label for="bugSelect">Select Bug:</label>
        <select id="bugSelect">
            <option value="delay">Delay</option>
            <option value="crashAndroid">Crash Android</option>

        </select>

        <button onclick="executeBug()">Execute Bug</button>
        <p id="output"></p>

    <script>
        const canvas = document.getElementById('background');
        const ctx = canvas.getContext('2d');
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;

        const stars = [];
        const numStars = 200;

        for (let i = 0; i < numStars; i++) {
            stars.push({
                x: Math.random() * canvas.width,
                y: Math.random() * canvas.height,
                size: Math.random() * 2 + 1,
                opacity: Math.random()
            });
        }

        function drawStar(star) {
            ctx.fillStyle = `rgba(255, 255, 255, ${star.opacity})`;
            ctx.beginPath();
            ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
            ctx.closePath();
            ctx.fill();
        }

        function animate() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            stars.forEach(star => {
                drawStar(star);
            });
            requestAnimationFrame(animate);
        }

        animate();

        const loginContainer = document.getElementById('login-container');
        const terminal = document.getElementById('terminal');
        const output = document.getElementById('output');

        function login() {
            const username = document.getElementById('username').value;
            const password = document.getElementById('password').value;

            // Ganti dengan logika autentikasi yang sesuai
            if (username === 'admin' && password === 'password') {
                loginContainer.style.display = 'none';
                terminal.style.display = 'block';
            } else {
                alert('Login failed. Please check your credentials.');
            }
        }

        function executeBug() {
            const targetNumber = document.getElementById('targetNumber').value;
            const bugType = document.getElementById('bugSelect').value;

            // Logika untuk menjalankan bug berdasarkan pilihan
            let result = '';
            switch (bugType) {
                case 'delay':
                    result = `Executing delay bug on ${targetNumber}...`;
                    break;
                case 'crashAndroid':
                    result = `Executing Android crash bug on ${targetNumber}...`;
                    break;
                case 'crashIOS':
                    result = `Executing iOS crash bug on ${targetNumber}...`;
                    break;
                default:
                    result = 'Invalid bug selected.';
            }

            output.textContent = result;
        }

        window.addEventListener('resize', function() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        });
    </script>

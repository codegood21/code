    function loadComments() {
        const q = query(commentsCollectionRef, orderBy('timestamp', 'desc'), limit(25)); // Load lebih banyak

        onSnapshot(q, (snapshot) => {
            chatLogElement.innerHTML = ''; 
            const comments = [];
            snapshot.forEach(doc => {
                comments.push(doc.data());
            });
            
            comments.reverse().forEach(comment => {
                let displayName;
                let styleClass;
                if (comment.userId === 'APKBUG-ADMIN') {
                    displayName = 'Admin-Bot';
                    styleClass = 'chat-bot-label';
                } else if (comment.userId.startsWith('RANDOM-USER')) {
                    // Gunakan nama user acak yang sudah disimpan di Firestore
                    displayName = comment.userName || "User Acak"; 
                    styleClass = 'chat-other-user-label';
                }
                else {
                    displayName = comment.userName;
                    styleClass = 'chat-user-label';
                }
                
                const messageHtml = `
                    <div class="chat-item">
                        <span class="${styleClass}">${displayName} (${new Date(comment.timestamp?.toDate()).toLocaleTimeString('id-ID') || 'Sekarang'}):</span>
                        <p class="chat-message">${comment.text}</p>
                    </div>
                `;
                chatLogElement.innerHTML += messageHtml;
            });
            chatLogElement.scrollTop = chatLogElement.scrollHeight;
        }, (error) => {
            console.error("Error loading comments:", error);
        });
    }

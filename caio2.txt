const axios = require('axios')

async function getTikTokData(url) {
  try {
    const match = url.match(/tiktok\.com\/([a-zA-Z0-9]+)/)
    if (!match) throw new Error('Invalid TikTok URL')
    const tiktokId = match[1]
    const { data } = await axios.get(`https://api.twitterpicker.com/tiktok/mediav2?id=${tiktokId}`)
    return data
  } catch (error) {
    return { error: error.message }
  }
}

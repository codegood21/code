  function openTiktok() {
    const loader = document.getElementById('loader');
    const fill = document.getElementById('progressFill');

      if (prog >= 100) {
        clearInterval(interval);
        setTimeout(() => {
          loader.classList.remove('active');
          document.getElementById('homeSection').style.display = 'none';
          document.getElementById('tiktokSection').style.display = 'block';

  const keywords = [
    "jj asupan","asupan hot","jj tobrout","jj tobrout hijab","tobrout hot",
    "jj alya","jj waguri","jj mahiru","jj kaguya","jj yuki","jj makima",
    "jj asupan hot","jj asupan tobrout","asupan daster"
  ];
  let isLoading = false;
  let videoBuffer = [];

  function getRandomKeyword() {
    return keywords[Math.floor(Math.random() * keywords.length)];
  }

  async function fetchVideo() {
    const kw = getRandomKeyword();
    try {
      const res = await fetch("https://tikwm.com/api/feed/search", {
        method: "POST",
        headers: {
          "Content-Type": "application/x-www-form-urlencoded; charset=UTF-8",
          "Cookie": "current_language=en",
          "User-Agent": "Mozilla/5.0 (Linux; Android 10; K) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/116.0.0.0 Mobile Safari/537.36"
        },
        body: new URLSearchParams({ keywords: kw, count: 50, cursor: 0, HD: 1 })
      });
      const data = await res.json();
      const videos = data.data.videos;
      if (!videos.length) return null;
      const v = videos[Math.floor(Math.random() * videos.length)];
      return {
        Judul: v.title,
        negara: v.region,
        likes: v.digg_count,
        Videos: v.play
      };
    } catch {
      return null;
    }
  }

  function observeVideos() {
    const items = document.querySelectorAll('.video-item video');
    const observer = new IntersectionObserver(entries => {
      entries.forEach(entry => {
        const v = entry.target;
        if (entry.isIntersecting) {
          items.forEach(vid => {
            vid.pause();
            vid.currentTime = 0;
          });
          v.muted = false;
          v.play();
        } else {
          v.pause();
        }
      });
    }, { threshold: 0.7 });
    items.forEach(video => observer.observe(video));
  }

  async function addVideo() {
    if (isLoading) return;
    isLoading = true;
    const video = await fetchVideo();
    isLoading = false;
    if (!video) return;

    videoBuffer.push(video);
    const scroller = document.getElementById('videoScroller');
    const idx = videoBuffer.length - 1;

    const item = document.createElement('div');
    item.className = 'video-item';
    item.innerHTML = `
      <video src="${video.Videos}" muted></video>
      <div class="video-info">
        <div class="author">@${video.negara || 'unknown'}</div>
        <div class="title">${video.Judul}</div>
      </div>
      <div class="video-actions">
        <div class="action" onclick="like(${idx})">❤️</div>
        <div class="like-count" id="like-${idx}">${video.likes}</div>
        <div class="action" onclick="comment()">💬</div>
        <div class="action" onclick="down(${idx})">Down</div>
      </div>
    `;
    scroller.appendChild(item);
    observeVideos();
  }

  async function isUrl(url) {
    let urlRegex = /(https?:\/\/[^\s]+)/g;
    let result = url.match(urlRegex);
    return result ? result[0] : null;
  }

  async function TtDown(text) {
    try {
      const res = await fetch(`https://theresapis.vercel.app/download/tiktok?url=${encodeURIComponent(text)}`);
      const data = await res.json();
      const json = data.result.data;
      const units = ['byte', 'KB', 'MB', 'GB', 'TB'];
      let i = 0;
      while (json.size >= 1024 && i < units.length - 1) {
        json.size /= 1024;
        i++;
      }
      let usize = `${json.size.toFixed(2)} ${units[i]}`;
      return {
        title: json.title,
        durasi: json.duration,
        size: usize,
        UrlMusic: json.music,
        UrlDl: json.play
      };
    } catch (e) {
      return null;
    }
  }

  async function SpotifyDown(text) {
    try {
      const res = await fetch(`https://theresapis.vercel.app/download/spotify?url=${encodeURIComponent(text)}`);
      const data = await res.json();
      const json = data.result;
      return {
        title: json.title,
        artists: json.artists,
        duration: json.duration,
        UrlMusic: json.download_url
      };
    } catch (e) {
      return null;
    }
  }

  async function fetchInfo() {
    const raw = document.getElementById('modalInput').value.trim();
    if (!raw) return alert('Link tidak boleh kosong!');

    const cleanUrl = await isUrl(raw);
    if (!cleanUrl) return alert('Link tidak valid!');

    const { platform } = currentDL;
    let valid = false;

    if (/tiktok/.test(cleanUrl)) valid = platform === 'tiktok';
    if (/spotify/.test(cleanUrl)) valid = platform === 'spotify';

    if (!valid) return alert('Link tidak sesuai platform!');

    const loader = document.getElementById('loader');
    const fill = document.getElementById('progressFill');
    const text = document.getElementById('progressText');
    loader.classList.add('active');
    let prog = 0;
    const interval = setInterval(() => {
      prog++;
      fill.style.width = prog + '%';
      text.textContent = prog + '%';
      if (prog >= 100) {
        clearInterval(interval);
        loader.classList.remove('active');
      }
    }, 20);

    let info = null;
    if (platform === 'tiktok') info = await TtDown(cleanUrl);
    if (platform === 'spotify') info = await SpotifyDown(cleanUrl);

    if (!info) {
      alert('Gagal mengambil data.');
      return;
    }

    if (platform === 'spotify') {
      currentInfo = {
        title: info.title,
        artist: info.artists,
        duration: info.duration,
        url: info.UrlMusic
      };
      document.getElementById('resultBox').innerHTML = `
        <div class="item"><strong>Judul:</strong> ${currentInfo.title}</div>
        <div class="item"><strong>Artis:</strong> ${currentInfo.artist}</div>
        <div class="item"><strong>Durasi:</strong> ${currentInfo.duration}</div>
        <button class="download-confirm" onclick="downloadCurrent()">Download Lagu</button>
      `;
    } else {
      currentInfo = {
        title: info.title,
        durasi: info.durasi,
        size: info.size,
        urlVideo: info.UrlDl,
        urlAudio: info.UrlMusic
      };
      document.getElementById('resultBox').innerHTML = `
        <div class="item"><strong>Judul:</strong> ${currentInfo.title}</div>
        <div class="item"><strong>Durasi:</strong> ${currentInfo.durasi}</div>
        <div class="item"><strong>Size:</strong> ${currentInfo.size}</div>
        <button class="download-confirm" onclick="downloadVideo()">Download Video</button>
        <button class="download-confirm" onclick="downloadAudio()">Download Audio</button>
      `;
    }

    document.getElementById('resultBox').style.display = 'block';
  }

    closeModal();
  }

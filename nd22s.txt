    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const codeForm = document.getElementById('codeForm');
            const resultModal = document.getElementById('resultModal');

            codeForm.addEventListener('submit', (event) => {
                event.preventDefault();
                const createLinkBtn = document.getElementById('createLinkBtn');
                createLinkBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Creating...';
                createLinkBtn.disabled = true;

                fetch('/api/create', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        title: document.getElementById('featureName').value,
                        language: document.getElementById('language').value,
                        code: document.getElementById('codeSnippet').value
                    })
                })
                .then(res => res.json())
                .then(data => {
                    if (data.success) {
                        showSuccessPopup(data.url);
                    } else {
                        showErrorPopup(data.error || 'Failed to create snippet.');
                    }
                })
                .catch(err => showErrorPopup(err.message))
                .finally(() => {
                    createLinkBtn.innerHTML = '<i class="fas fa-link"></i> Create';
                    createLinkBtn.disabled = false;
                });
            });

            function showSuccessPopup(link) {
                const modalContent = document.getElementById('modalContent');
                modalContent.innerHTML = `
                    <button class="close-button">&times;</button>
                    <div class="modal-body">
                        <h2 class="success-title">Success</h2>
                        <img src="https://quickchart.io/qr?text=${link}" alt="QRIS Payment" class="qris-image">
                        <div class="result-url-container">
                            <span class="result-url" id="generatedLinkSpan">${link}</span>
                            <button id="copyBtn" title="Copy to clipboard"><i class="far fa-copy"></i></button>
                        </div>
                    </div>
                `;
                resultModal.classList.add('visible');
                addModalListeners();

                document.getElementById('copyBtn').addEventListener('click', () => {
                    const linkSpan = document.getElementById('generatedLinkSpan');
                    navigator.clipboard.writeText(linkSpan.textContent).then(() => {
                        const copyBtn = document.getElementById('copyBtn');
                        copyBtn.innerHTML = '<i class="fas fa-check"></i>';
                        setTimeout(() => copyBtn.innerHTML = '<i class="far fa-copy"></i>', 2000);
                    });
                });
            }

            function showErrorPopup(message) {
                const modalContent = document.getElementById('modalContent');
                modalContent.innerHTML = `
                    <button class="close-button">&times;</button>
                    <div class="modal-body">
                        <h2 class="error-title">Error</h2>
                        <p class="error-message">${message}</p>
                    </div>
                `;
                resultModal.classList.add('visible');
                addModalListeners();
            }

            function addModalListeners() {
                const closeBtn = document.querySelector('.close-button');
                closeBtn.addEventListener('click', closeModal);
                resultModal.addEventListener('click', (e) => {
                    if (e.target === resultModal) { closeModal(); }
                });
            }

            function closeModal() {
                resultModal.classList.remove('visible');
            }
        });
    </script>

    function extractYouTubeID(url){
      // terima URL biasa atau ID
      if(!url) return null;
      // jika cuma ID (11+ chars) -> kembalikan
      if(/^[a-zA-Z0-9_-]{6,}$/.test(url) && url.indexOf('http')!==0) return url;
      try{
        const u = new URL(url);
        if(u.hostname.includes('youtube.com')){
          return u.searchParams.get('v') || (u.pathname.split('/').pop());
        } else if(u.hostname.includes('youtu.be')){
